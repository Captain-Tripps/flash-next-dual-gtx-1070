/* ------------------------------------------------------------------ *
 * Voxel ids -> base colour. One entry per surface in the scene.
 * A block's final colour is its base, modulated by baked ambient
 * occlusion plus a per-block tint, so the same material still reads
 * as a pile of individual cubes rather than one smooth mass.
 * ------------------------------------------------------------------ */

const T = {
  AIR: 0, ROCK: 1, SAND: 2, WET: 3, GRASS: 4, MOSS: 5, HEWN: 6, PAVE: 7,

};

const PALETTE = {

/* Self-lit blocks get their own merged group so the solid geometry can
 * stay one cheap MeshStandardMaterial with vertex colours. */
const EMISSIVE_IDS = new Set([T.EMBER, T.FLAME, T.EYE]);

/* Cliff rock: every third y-band is tone-shifted, so the cliff face reads
 * as strata instead of one flat grey slab. */
function strataTint(y) {
  const ramp = [1.0, 0.93, 1.04];
  const band = ((Math.floor(y / 3) % 3) + 3) % 3;
  return ramp[band];
}

/* ------------------------------------------------------------------ *
 * Deterministic noise. Deliberate, not random: one seed lays the whole
 * world, so a reload never rebuilds the stack differently on you.
 * ------------------------------------------------------------------ */

function hash2(x, y) {
  const s = Math.sin(x * 127.1 + y * 311.7) * 43758.5453;
  return s - Math.floor(s);
}
function hash3(x, y, z) {
  const s = Math.sin(x * 127.1 + y * 311.7 + z * 74.7) * 43758.5453;
  return s - Math.floor(s);
}

/* ------------------------------------------------------------------ *
 * Two containers, one builder API. Field is a bounded dense grid for the
 * world; Bag is unbounded and sparse, used for dragons and eggs, which
 * are placed as units and therefore need no world bounds.
 * ------------------------------------------------------------------ */

const Shapes = {
  fill(x, y, z, type) { return this.set(x, y, z, type); },

  box(x0, x1, y0, y1, z0, z1, type) {
    for (let z = z0; z <= z1; z++)
      for (let y = y0; y <= y1; y++)
        for (let x = x0; x <= x1; x++) this.set(x, y, z, type);
    return this;
  },

  /* the same box, spelled from two corner points */
  block([ax, ay, az], [bx, by, bz], type) {
    return this.box(Math.min(ax, bx), Math.max(ax, bx), Math.min(ay, by),
      Math.max(ay, by), Math.min(az, bz), Math.max(az, bz), type);
  },

  column(x, z, y0, y1, type) {
    for (let y = Math.min(y0, y1); y <= Math.max(y0, y1); y++) this.set(x, y, z, type);
    return this;
  },

  /* a rope of blocks between two points; gap=2 for slack rope, 1 for taut */
  strand(x0, y0, z0, x1, y1, z1, type, gap = 1) {
    const dx = x1 - x0, dy = y1 - y0, dz = z1 - z0;
    const steps = Math.max(1, Math.ceil(Math.max(Math.abs(dx), Math.abs(dy), Math.abs(dz)) / gap));
    for (let i = 0; i <= steps; i++) {
      this.set(
        Math.round(x0 + dx * i / steps),
        Math.round(y0 + dy * i / steps),
        Math.round(z0 + dz * i / steps),
        type
      );
    }
    return this;
  },

  /* a blocky ellipsoid: diam is the inscribed box width; a voxel is solid
   * when its centre falls inside that ellipsoid. Odd diam = exact width. */
  blob(cx, cy, cz, rx, ry, rz, type) {
    for (let dz = -Math.ceil(rz); dz <= Math.ceil(rz); dz++)
      for (let dy = -Math.ceil(ry); dy <= Math.ceil(ry); dy++)
        for (let dx = -Math.ceil(rx); dx <= Math.ceil(rx); dx++) {
          const q = (dx * dx) / (rx * rx) + (dy * dy) / (ry * ry) + (dz * dz) / (rz * rz);
          if (q <= 1) this.set(cx + dx, cy + dy, cz + dz, type);
        }
    return this;
  },

  /* a corbelled dome: ring n shrinks by 1 each lift, sampled at voxel centres */
  cupola(cx, cy, cz, rx, rz, lifts, type) {
    for (let lift = 0; lift < lifts; lift++) {
      const f = 1 - lift / lifts;
      const limX = Math.ceil(rx * f), limZ = Math.ceil(rz * f);
      for (let dz = -limZ; dz <= limZ; dz++)
        for (let dx = -limX; dx <= limX; dx++) {
          const ox = dx + 0.5, oz = dz + 0.5;
          if ((ox * ox) / (rx * rx) + (oz * oz) / (rz * rz) <= (rx * f) * (rx * f)) {
            this.set(cx + dx, cy + lift, cz + dz, type);
          }
        }
    }
    return this;
  },

  /* hollow a box back out to air; set() refuses air, so carving is its own call */
  carve(x0, x1, y0, y1, z0, z1) {
    for (let z = z0; z <= z1; z++)
      for (let y = y0; y <= y1; y++)
        for (let x = x0; x <= x1; x++) this.clear(x, y, z);
    return this;
  },

  /* perimeter of a rectangle on one lift, for parapets and railings */
  rim(x0, x1, z0, z1, y, type) {
    for (let x = x0; x <= x1; x++) { this.set(x, y, z0, type); this.set(x, y, z1, type); }
    for (let z = z0; z <= z1; z++) { this.set(x0, y, z, type); this.set(x1, y, z, type); }
    return this;
  },
};

class Field {
  constructor(minX, minY, minZ, sizeX, sizeY, sizeZ) {
    this.min = [minX, minY, minZ];
    this.size = [sizeX, sizeY, sizeZ];
    this.data = new Uint8Array(sizeX * sizeY * sizeZ);
  }
  index(x, y, z) {
    const [mx, my, mz] = this.min, sx = this.size[0], sy = this.size[1];
    return (x - mx) + sx * ((y - my) + sy * (z - mz));
  }
  inside(x, y, z) {
    const [mx, my, mz] = this.min, [sx, sy, sz] = this.size;
    return x >= mx && x < mx + sx && y >= my && y < my + sy && z >= mz && z < mz + sz;
  }
  get(x, y, z) { return this.inside(x, y, z) ? this.data[this.index(x, y, z)] : T.AIR; }
  set(x, y, z, type) {
    if (!this.inside(x, y, z) || type === T.AIR) return this;
    this.data[this.index(x, y, z)] = type;
    return this;
  }
  clear(x, y, z) {
    if (this.inside(x, y, z)) this.data[this.index(x, y, z)] = T.AIR;
    return this;
  }
  cells() {
    const out = [], [mx, my, mz] = this.min, [sx, sy, sz] = this.size;
    for (let z = mz; z < mz + sz; z++)
      for (let y = my; y < my + sy; y++)
        for (let x = mx; x < mx + sx; x++) {
          const type = this.get(x, y, z);
          if (type) out.push([x, y, z, type]);
        }
    return out;
  }
}

class Bag {
  constructor() { this.map = new Map(); }
  /* Base-1801 packing of (x,y,z) into one Map key so a dragon shape can live in
   * a sparse grid without string keys. Range is +-900 around an origin. */
  static key(x, y, z) { return (x + 900) * 1801 * 1801 + (y + 900) * 1801 + (z + 900); }
  static unkey(k) {
    const z = (k % 1801) - 900;
    const t = Math.floor(k / 1801);
    const y = (t % 1801) - 900;
    const x = Math.floor(t / 1801) - 900;
    return [x, y, z];
  }
  get(x, y, z) { return this.map.get(Bag.key(x, y, z)) || T.AIR; }
  clear(x, y, z) { this.map.delete(Bag.key(x, y, z)); return this; }
  set(x, y, z, type) {
    if (type === T.AIR) this.map.delete(Bag.key(x, y, z));
    else this.map.set(Bag.key(x, y, z), type);
    return this;
  }
  cells() {
    const out = [];
    for (const [k, type] of this.map) out.push([...Bag.unkey(k), type]);
    return out;
  }
}

Object.assign(Field.prototype, Shapes);
